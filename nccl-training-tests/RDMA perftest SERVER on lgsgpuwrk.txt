# RDMA perftest SERVER on lgsgpuwrk001 (backend port ens1f1np1). Static IP 10.132.200.2/16.
#   oc apply -f rdma-qos2-lgsgpuwrk001.yaml
# master must be a BACKEND ConnectX-8 netdev that exists on this node (confirm: ibdev2netdev).
apiVersion: k8s.cni.cncf.io/v1
kind: NetworkAttachmentDefinition
metadata:
  name: rdma-qos2-srv-runai
  namespace: runai-cisco-team
spec:
  config: '{ "cniVersion":"0.3.1", "name":"rdma-qos2-srv-net", "type":"macvlan", "master":"ens1f1np1", "mode":"bridge", "mtu":9000, "ipam":{ "type":"static", "addresses":[ {"address":"10.132.200.2/16"} ] } }'
---
apiVersion: v1
kind: Pod
metadata:
  name: rdma-qos2-lgsgpuwrk001
  namespace: runai-cisco-team
  labels:
    app: rdma-qos-test2
    role: server
  annotations:
    k8s.v1.cni.cncf.io/networks: '[{"name":"rdma-qos2-srv-runai","namespace":"runai-cisco-team"}]'
spec:
  nodeSelector:
    kubernetes.io/hostname: lgsgpuwrk001.lgsgpu.svc.cvshealth.com
  restartPolicy: Always
  containers:
  - name: rdma
    image: quay.io/edge-infrastructure/nvidia-tools:0.1.5
    securityContext:
      capabilities:
        add: ["IPC_LOCK","SYS_RESOURCE","NET_RAW","NET_ADMIN"]
    resources:
      limits:
        rdma/rdma_shared_device_a: 1
    command: ["bash","-c"]
    args:
    - |
      set -u; PORT=18516; SERVER_IP=10.132.200.2
      if ! command -v ib_write_bw >/dev/null 2>&1; then
        printf '[rk-app]\nname=rk-app\nbaseurl=http://dl.rockylinux.org/pub/rocky/9/AppStream/x86_64/os/\nenabled=1\ngpgcheck=0\n[rk-base]\nname=rk-base\nbaseurl=http://dl.rockylinux.org/pub/rocky/9/BaseOS/x86_64/os/\nenabled=1\ngpgcheck=0\n' > /etc/yum.repos.d/rk.repo
        dnf -y --disablerepo='*' --enablerepo='rk-app,rk-base' install perftest; unset http_proxy https_proxy
      fi
      until ip -4 addr show net1 2>/dev/null | grep -q "$SERVER_IP"; do echo "waiting net1 $SERVER_IP"; sleep 2; done
      DEV=$(show_gids | awk -v ip="$SERVER_IP" '$5==ip && $6=="v2"{print $1; exit}')
      echo "SERVER ip=$SERVER_IP dev=$DEV port=$PORT"
      while true; do ib_write_bw -d "$DEV" -R -F --report_gbits -q 2 -b -p "$PORT" --run_infinitely || true; sleep 3; done
